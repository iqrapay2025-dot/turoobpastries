function Carousel({ children, cardW }: { children: React.ReactNode[]; cardW: number }) {
  const ref = useRef<HTMLDivElement>(null)
  const [i, setI] = useState(0)
  const n = children.length
  const go = (to: number) => {
    const el = ref.current
    if (!el) return
    const k = Math.max(0, Math.min(n - 1, to))
    el.scrollTo({ left: k * (cardW + 16), behavior: 'smooth' })
    setI(k)
  }
  return (
    <div>
      <div ref={ref} onScroll={(e) => setI(Math.round(e.currentTarget.scrollLeft / (cardW + 16)))} className="no-scrollbar -mx-4 flex snap-x snap-mandatory gap-4 overflow-x-auto px-4 pb-6 sm:mx-0 sm:px-0">
        {children}
      </div>
      <div className="flex items-center justify-between">
        <div>
          <p className="text-sm font-semibold text-ink-soft sm:hidden" aria-live="polite">{i + 1} / {n}</p>
          <div className="hidden sm:flex">
            {children.map((_, k) => (
              <button key={k} aria-label={`Go to slide ${k + 1}`} aria-current={k === i} onClick={() => go(k)} className="group/dot grid size-11 place-items-center rounded-full">
                <span className={'block h-2.5 rounded-full transition-all ' + (k === i ? 'w-8 bg-red' : 'w-2.5 bg-red-light group-hover/dot:bg-red')} />
              </button>
            ))}
          </div>
        </div>
        <div className="flex gap-2">
          <button aria-label="Previous" disabled={i === 0} onClick={() => go(i - 1)} className="grid size-12 place-items-center rounded-full bg-white shadow-soft transition hover:-translate-x-0.5 hover:bg-yellow active:scale-95 disabled:opacity-40 disabled:hover:translate-x-0 disabled:hover:bg-white"><ChevronLeft /></button>
          <button aria-label="Next" disabled={i === n - 1} onClick={() => go(i + 1)} className="grid size-12 place-items-center rounded-full bg-red text-white shadow-soft transition hover:translate-x-0.5 hover:bg-red-dark active:scale-95 disabled:opacity-40 disabled:hover:translate-x-0 disabled:hover:bg-red"><ChevronRight /></button>
        </div>
      </div>
    </div>
  )
}
